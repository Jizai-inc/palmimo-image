# Source packages excluded from /usr/share/palmimo/sources/ (see
# lib/collect_oss_compliance.py and README.md, "Corresponding source for
# the rest of the image"). One source package name per line, `#` for
# comments/reasons. Every other apt source package on the image IS
# collected -- this list is meant to stay short and hand-reviewed: apt's
# own license metadata is not trustworthy enough to filter by license
# family, so it is not used to decide what to skip.
#
# Each entry below is a non-free firmware blob that carries no
# source-provision obligation (no GPL/LGPL component requiring source, just
# a binary redistribution license) -- there is nothing to collect even in
# principle.

# firmware-brcm80211, firmware-atheros, firmware-libertas, firmware-mediatek,
# firmware-realtek, etc. are all built from this one source package.
firmware-nonfree

# Raspberry Pi boot firmware (start*.elf, bootcode.bin, GPU firmware) --
# Debian does carry a raspi-firmware source package, but its contents are
# binary blobs under Broadcom's redistribution license with no GPL/LGPL
# component inside that would trigger a source-provision obligation.
raspi-firmware

# Bluetooth controller firmware blobs (brcm/bcm43xx .hcd files) --
# binary-only, same rationale as firmware-nonfree.
bluez-firmware
